# 1.认识时钟(F1)
![Pasted image 20231215143526](../../../../uploads/stm32-hal-library/pasted-image-20231215143526.png)

|时钟源名称|频率|材料|用途|
|---|---|---|---|
|高速外部振荡器(HSE)|4~16MHz|晶体/陶瓷|SYSCLK/RTC|
|低速外部振荡器(LSE)|32.768KHz|晶体/陶瓷|RTC|
|高速内部振荡器(HSI)|8MHz|RC|SYSCLK|
|低速内部振荡器(LSI)|40KHz|RC|RTC/IWDG|

# 2.配置步骤
1. 选择时钟源，配置PLL
通过HAL_RCC_OscConfig()函数设置
2. 选择系统时钟源，配置总线分频器
   通过HAL_RCC_ClockConfig()函数设置


# 3.系统初始化函数
1.HAL_RCC_OscConfig(RCC_OscInitTypeDef * RCC_OscInitStruct)函数
```c
typedef struct
{
  uint32_t  OscillatorType;   /* 选择需要配置的振荡器 */
  uint32_t  HSEState;   /* HSE 状态 */
  uint32_t  HSEPredivValue;   /* HSE 预分频值 */
  uint32_t  LSEState;   /* LSE 状态 */
  uint32_t  HSIState;   /* HSI状态 */
  uint32_t  HSICalibrationValue;   /* HSI 校准值 */
  uint32_t  LSIState;   /* LSI 状态 */
  RCC_PLLInitTypeDef  PLL;   /* PLL 结构体 */
}RCC_OscInitTypeDef;

typedef struct
{
  uint32_t  PLLState;   /* PLL 状态 */
  uint32_t  PLLSource;   /* PLL 时钟源 */
  uint32_t  PLLMUL;   /* PLL 倍频系数 */
}RCC_PLLInitTypeDef;
```
2.HAL_RCC_ClockConfig(RCC_ClkInitTypeDef * RCC_ClkInitStruct, uint32_t FLatency)函数
```c
typedef struct
{
  uint32_t  ClockType;   /* 要配置的时钟（SYSCLK/HCLK/PCLK1/PCLK2） */
  uint32_t  SYSCLKSource;   /* 系统时钟源 */
  uint32_t  AHBCLKDivider;   /* AHB  时钟预分频系数 */
  uint32_t  APB1CLKDivider;   /* APB1 时钟预分频系数 */
  uint32_t  APB2CLKDivider;   /* APB2 时钟预分频系数 */
}RCC_ClkInitTypeDef;

uint32_t FLatency
#define  FLASH_LATENCY_0   0x00000000U   /* FLASH 0个等待周期 */
#define  FLASH_LATENCY_1   FLASH_ACR_LATENCY_0 /* FLASH 1个等待周期 */
#define  FLASH_LATENCY_2   FLASH_ACR_LATENCY_1   /* FLASH 2个等待周期 */
```
在STM32的系统配置中，Flash延迟是一个重要的参数。当系统时钟频率提高时，Flash的读取速度可能无法跟上，这时就需要增加Flash延迟，也就是CPU读取Flash的等待周期。例如，对于STM32F10x系列，当系统时钟频率在24MHz以下时，Flash延迟可以设置为0；当系统时钟频率在24MHz到48MHz之间时，Flash延迟需要设置为1；当系统时钟频率在48MHz到72MHz之间时，Flash延迟需要设置为2。
# 4.示例代码
```c
void RCC_Clock_Init()

{

    RCC_OscInitTypeDef RCC_OscInitStruct;
    RCC_ClkInitTypeDef RCC_ClkInitStruct;

    // 设置振荡器类型为HSE（高速外部）
    RCC_OscInitStruct.OscillatorType=RCC_OSCILLATORTYPE_HSE;

    // 启用HSE
    RCC_OscInitStruct.HSEState=RCC_HSE_ON;

    // 设置HSE预分频值为1，这意味着HSE频率不变，仍为8MHz
    RCC_OscInitStruct.HSEPredivValue=RCC_HSE_PREDIV_DIV1;

    // 启用PLL（相位锁定环）
    RCC_OscInitStruct.PLL.PLLState=RCC_PLL_ON;

    // 将PLL源设置为HSE
    RCC_OscInitStruct.PLL.PLLSource=RCC_PLLSOURCE_HSE;

    // 将PLL倍频设置为9，这意味着PLL输出频率为8MHz * 9 = 72MHz
    RCC_OscInitStruct.PLL.PLLMUL=RCC_PLL_MUL9;

    // 配置振荡器参数
    HAL_RCC_OscConfig(&RCC_OscInitStruct);

    // 设置时钟类型为系统时钟（SYSCLK）、高速总线时钟（HCLK）、低速外设时钟1（PCLK1）和低速外设时钟2（PCLK2）
    RCC_ClkInitStruct.ClockType=RCC_CLOCKTYPE_SYSCLK |RCC_CLOCKTYPE_HCLK |RCC_CLOCKTYPE_PCLK1 |RCC_CLOCKTYPE_PCLK2 ;

    // 将系统时钟源设置为PLL时钟，即72MHz
    RCC_ClkInitStruct.SYSCLKSource=RCC_SYSCLKSOURCE_PLLCLK;

    // 将AHB时钟分频器设置为1，这意味着AHB时钟频率为72MHz
    RCC_ClkInitStruct.AHBCLKDivider=RCC_SYSCLK_DIV1;

    // 将APB1时钟分频器设置为2，这意味着APB1时钟频率为72MHz / 2 = 36MHz
    RCC_ClkInitStruct.APB1CLKDivider=RCC_HCLK_DIV2;

    // 将APB2时钟分频器设置为1，这意味着APB2时钟频率为72MHz
    RCC_ClkInitStruct.APB2CLKDivider=RCC_HCLK_DIV1;

    // 配置时钟参数，并设置Flash延迟为2
    HAL_RCC_ClockConfig(&RCC_ClkInitStruct,FLASH_LATENCY_2);
}
```
